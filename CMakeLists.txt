cmake_minimum_required(VERSION 3.24)
set(GUILD_PACKAGE_VERSION "1.1.1" CACHE STRING "Semantic package release version")
project(guild_kernels_cpu VERSION ${GUILD_PACKAGE_VERSION} LANGUAGES C CXX)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  OUTPUT_VARIABLE PACKAGE_SOURCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
find_package(GuildKernelSDK 1.3 REQUIRED CONFIG)
option(GUILD_CPU_GGML_AVX512 "Use GGML AVX-512 arithmetic; package requires AVX512F/BW/DQ/VL" OFF)
include(FetchContent)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_STATIC ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
foreach(feature AVX AVX2 FMA F16C BMI2)
  set(GGML_${feature} ON CACHE BOOL "" FORCE)
endforeach()
foreach(feature AVX512 AVX512_VBMI AVX512_VNNI AVX_VNNI AVX512_BF16 AMX_TILE AMX_INT8 AMX_BF16)
  set(GGML_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
if(GUILD_CPU_GGML_AVX512)
  set(GGML_AVX512 ON CACHE BOOL "" FORCE)
  set(CPU_REQUIRED_ISA 5116) # AVX2/FMA/F16C/BMI2 + AVX512F/BW/DQ/VL/CD
else()
  set(CPU_REQUIRED_ISA 60) # GGML is also compiled with BMI2
endif()
configure_file(package.json.in ${CMAKE_CURRENT_BINARY_DIR}/package.json @ONLY)
FetchContent_Declare(llama GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
  GIT_TAG 3cf03257f219afbe7334045ff7c6a06ac68c627d GIT_SHALLOW FALSE)
FetchContent_GetProperties(llama)
if(NOT llama_POPULATED)
  FetchContent_Populate(llama)
endif()
add_subdirectory(${llama_SOURCE_DIR}/ggml ${CMAKE_BINARY_DIR}/ggml EXCLUDE_FROM_ALL)
if(GUILD_CPU_GGML_AVX512 AND NOT MSVC)
  # Pin the tuning/ISA contract instead of depending on the build host's native CPU.
  target_compile_options(ggml-cpu PRIVATE -march=skylake-avx512)
endif()
include(cmake/GuildCPUSources.cmake)
add_library(cpu_arithmetic STATIC)
guild_cpu_attach_base(cpu_arithmetic TRUE)
guild_cpu_attach_native(cpu_arithmetic)
target_compile_features(cpu_arithmetic PUBLIC cxx_std_17)
target_link_libraries(cpu_arithmetic PUBLIC Guild::KernelSDK ggml-cpu ggml-base)
add_library(guild_cpu_plugin SHARED src/plugin.cpp)
guild_configure_gkernel_plugin(guild_cpu_plugin)
target_link_libraries(guild_cpu_plugin PRIVATE cpu_arithmetic)
target_compile_definitions(guild_cpu_plugin PRIVATE GUILD_PACKAGE_VERSION="${PROJECT_VERSION}" CPU_REQUIRED_ISA=${CPU_REQUIRED_ISA}
  GUILD_CPU_VERSION_MAJOR=${PROJECT_VERSION_MAJOR} GUILD_CPU_VERSION_MINOR=${PROJECT_VERSION_MINOR} GUILD_CPU_VERSION_PATCH=${PROJECT_VERSION_PATCH})
set_target_properties(cpu_arithmetic guild_cpu_plugin PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(UNIX)
  target_link_options(guild_cpu_plugin PRIVATE -Wl,--no-undefined -Wl,-Bsymbolic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(cpu_contract_test tests/cpu_test.cpp)
  target_link_libraries(cpu_contract_test PRIVATE cpu_arithmetic guild_cpu_plugin)
  add_test(NAME expert_arithmetic COMMAND cpu_contract_test)
  add_executable(full_backend_test tests/full_backend_test.cpp)
  target_link_libraries(full_backend_test PRIVATE Guild::KernelSDK guild_cpu_plugin)
  add_test(NAME full_cpu_contract COMMAND full_backend_test)
  add_library(c_contract OBJECT tests/c_contract.c)
  target_link_libraries(c_contract PRIVATE Guild::KernelSDK)
endif()
install(TARGETS guild_cpu_plugin LIBRARY DESTINATION lib RUNTIME DESTINATION lib)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/package.json DESTINATION . RENAME .gkmetadata)
